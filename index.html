function doGet(e) {
  var action = e.parameter.action;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action == "getData") {
    var studentSheet = getOrCreateSheet(ss, "Anggota", ["ID", "Nama", "Kelas"]);
    var attendanceSheet = getOrCreateSheet(ss, "Absensi", ["Waktu", "ID", "Nama", "Kelas", "Status"]);
    var configSheet = getOrCreateSheet(ss, "Pengaturan", ["Key", "Value"]);
    
    return ContentService.createTextOutput(JSON.stringify({
      students: getSheetData(studentSheet),
      attendance: getSheetData(attendanceSheet),
      config: getSheetData(configSheet)
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  var action = data.action;

  if (action == "addAttendance") {
    var sheet = getOrCreateSheet(ss, "Absensi", ["Waktu", "ID", "Nama", "Kelas", "Status"]);
    
    // Cek Pencegahan Absen Ganda Sisi Server (Google Sheets)
    var rows = sheet.getDataRange().getValues();
    var todayStr = new Date().toLocaleDateString("en-CA"); // Format YYYY-MM-DD
    
    for (var i = 1; i < rows.length; i++) {
      var rowId = String(rows[i][1]).trim().toLowerCase();
      var rowTime = rows[i][0];
      var rowDateStr = "";
      
      if (rowTime instanceof Date) {
        rowDateStr = Utilities.formatDate(rowTime, ss.getSpreadsheetTimeZone(), "yyyy-MM-dd");
      } else {
        rowDateStr = String(rowTime).split(" ")[0].split(",")[0];
      }

      if (rowId === String(data.id).trim().toLowerCase() && rowDateStr.indexOf(todayStr) !== -1) {
        return responseJSON({ status: "ignored", message: "Siswa sudah absen hari ini." });
      }
    }

    sheet.appendRow(["'" + data.timestamp, "'" + data.id, data.name, data.className, data.status]);
    return responseJSON({ status: "success", message: "Absen berhasil dicatat!" });
  }

  if (action == "updateStudents") {
    var sheet = getOrCreateSheet(ss, "Anggota", ["ID", "Nama", "Kelas"]);
    sheet.clearContents();
    sheet.appendRow(["ID", "Nama", "Kelas"]);
    data.students.forEach(function(s) {
      sheet.appendRow(["'" + s.id, s.name, s.className]);
    });
    return responseJSON({ status: "success", message: "Data anggota berhasil diperbarui!" });
  }

  if (action == "saveConfig") {
    var sheet = getOrCreateSheet(ss, "Pengaturan", ["Key", "Value"]);
    sheet.clearContents();
    sheet.appendRow(["Key", "Value"]);
    sheet.appendRow(["RedirectURL", data.redirectUrl || ""]);
    return responseJSON({ status: "success", message: "Pengaturan berhasil disimpan!" });
  }
}

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
  }
  return sheet;
}

function getSheetData(sheet) {
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0];
  var result = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = data[i][j];
    }
    result.push(obj);
  }
  return result;
}

function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
